# a. 0 ≡ λx. false, n + 1 ≡ pair true n
isNotZero = λn. n true

not = λb. b false true
isZero = λn. not (isNotZero n)

pred = λn. (isZero n) 0 (n false)

succ = λn. pair true n
suma = λn m. (isZero m) n (succ (suma n (pred m)))
suma = Y (λf n m. (isZero m) n (succ (f n (pred m))))
